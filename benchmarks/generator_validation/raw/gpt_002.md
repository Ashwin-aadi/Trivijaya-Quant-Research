## cross-sectional momentum

```python
from __future__ import annotations

from datetime import date

import polars as pl
from src.backtest.strategy import MarketView, Signal, Strategy


class CrossSectionalMomentum(Strategy):
    rationale = (
        "Stocks with the strongest intermediate-horizon returns often continue to outperform "
        "because information diffuses gradually and institutional investors adjust positions "
        "over time. Ranking securities cross-sectionally captures this persistence while "
        "avoiding dependence on the absolute market direction."
    )

    def __init__(self, lookback: int = 126, top_n: int = 10) -> None:
        self._lookback = lookback
        self._top_n = top_n

    def generate(self, view: MarketView) -> Signal:
        stamp = _latest_visible(view)
        closes = view.closes(lookback=self._lookback)
        if closes.height < self._lookback:
            return Signal(information_available_at=stamp, weights={})

        scores: list[tuple[str, float]] = []
        for symbol in view.symbols:
            if symbol not in closes.columns:
                continue
            prices = [float(x) for x in closes[symbol].drop_nulls().to_list()]
            if len(prices) < self._lookback:
                continue
            if prices[0] <= 0.0:
                continue
            scores.append((symbol, prices[-1] / prices[0] - 1.0))

        scores.sort(key=lambda x: x[1], reverse=True)
        picks = [s for s, _ in scores[: self._top_n]]
        if not picks:
            return Signal(information_available_at=stamp, weights={})

        w = 1.0 / len(picks)
        return Signal(
            information_available_at=stamp,
            weights={s: w for s in picks},
        )


def _latest_visible(view: MarketView) -> date:
    visible = view.history()
    if visible.is_empty():
        return date(1900, 1, 1)
    newest = visible["session_date"].max()
    assert isinstance(newest, date)
    return newest
```

## volatility-scaled trend following

```python
from __future__ import annotations

from datetime import date

import polars as pl
from src.backtest.strategy import MarketView, Signal, Strategy


class VolatilityScaledTrendFollowing(Strategy):
    rationale = (
        "Trend following exploits the tendency of persistent price moves to continue as "
        "investors gradually react to information. Scaling exposure by recent realized "
        "volatility reduces concentration in unstable names and improves risk balance."
    )

    def __init__(
        self,
        lookback: int = 63,
        top_n: int = 10,
        target_weight: float = 0.8,
    ) -> None:
        self._lookback = lookback
        self._top_n = top_n
        self._target_weight = target_weight

    def generate(self, view: MarketView) -> Signal:
        stamp = _latest_visible(view)
        closes = view.closes(lookback=self._lookback)
        if closes.height < self._lookback:
            return Signal(information_available_at=stamp, weights={})

        candidates: list[tuple[str, float]] = []
        for symbol in view.symbols:
            if symbol not in closes.columns:
                continue
            prices = [float(x) for x in closes[symbol].drop_nulls().to_list()]
            if len(prices) < self._lookback:
                continue
            if prices[0] <= 0.0:
                continue

            trend = prices[-1] / prices[0] - 1.0
            if trend <= 0.0:
                continue

            returns = []
            for i in range(1, len(prices)):
                if prices[i - 1] > 0.0:
                    returns.append(prices[i] / prices[i - 1] - 1.0)
            if len(returns) < 2:
                continue

            mean = sum(returns) / len(returns)
            var = sum((r - mean) ** 2 for r in returns) / (len(returns) - 1)
            vol = var ** 0.5
            if vol <= 0.0:
                continue

            candidates.append((symbol, 1.0 / vol))

        candidates.sort(key=lambda x: x[1], reverse=True)
        selected = candidates[: self._top_n]
        if not selected:
            return Signal(information_available_at=stamp, weights={})

        total = sum(v for _, v in selected)
        weights = {
            s: self._target_weight * inv_vol / total
            for s, inv_vol in selected
        }
        return Signal(information_available_at=stamp, weights=weights)


def _latest_visible(view: MarketView) -> date:
    visible = view.history()
    if visible.is_empty():
        return date(1900, 1, 1)
    newest = visible["session_date"].max()
    assert isinstance(newest, date)
    return newest
```

## liquidity-screened equal weighting

```python
from __future__ import annotations

from datetime import date

import polars as pl
from src.backtest.strategy import MarketView, Signal, Strategy


class LiquidityScreenedEqualWeight(Strategy):
    rationale = (
        "Equal weighting avoids concentration in the largest index constituents while "
        "screening on trading activity reduces implementation costs. Using recent average "
        "daily traded value helps ensure that selected names remain practically tradable."
    )

    def __init__(
        self,
        lookback: int = 20,
        top_n: int = 20,
    ) -> None:
        self._lookback = lookback
        self._top_n = top_n

    def generate(self, view: MarketView) -> Signal:
        stamp = _latest_visible(view)
        history = view.history(lookback=self._lookback)
        if history.is_empty():
            return Signal(information_available_at=stamp, weights={})

        liquidity: list[tuple[str, float]] = []
        for symbol in view.symbols:
            rows = history.filter(pl.col("symbol") == symbol)
            if rows.height < self._lookback:
                continue
            closes = [float(x) for x in rows["adj_close"].to_list()]
            volumes = [float(x) for x in rows["volume"].to_list()]
            traded = [c * v for c, v in zip(closes, volumes)]
            liquidity.append((symbol, sum(traded) / len(traded)))

        liquidity.sort(key=lambda x: x[1], reverse=True)
        picks = [s for s, _ in liquidity[: self._top_n]]
        if not picks:
            return Signal(information_available_at=stamp, weights={})

        weight = 1.0 / len(picks)
        return Signal(
            information_available_at=stamp,
            weights={s: weight for s in picks},
        )


def _latest_visible(view: MarketView) -> date:
    visible = view.history()
    if visible.is_empty():
        return date(1900, 1, 1)
    newest = visible["session_date"].max()
    assert isinstance(newest, date)
    return newest
```

## price-level reversion against a trailing reference

```python
from __future__ import annotations

from datetime import date

import polars as pl
from src.backtest.strategy import MarketView, Signal, Strategy


class TrailingReferenceReversion(Strategy):
    rationale = (
        "Large deviations below a trailing reference price can arise from temporary selling "
        "pressure rather than permanent changes in intrinsic value. Buying the names trading "
        "furthest below their recent average seeks to capture reversion once liquidity "
        "conditions normalize."
    )

    def __init__(self, window: int = 60, top_n: int = 10) -> None:
        self._window = window
        self._top_n = top_n

    def generate(self, view: MarketView) -> Signal:
        stamp = _latest_visible(view)
        closes = view.closes(lookback=self._window)
        if closes.height < self._window:
            return Signal(information_available_at=stamp, weights={})

        scores: list[tuple[str, float]] = []
        for symbol in view.symbols:
            if symbol not in closes.columns:
                continue
            prices = [float(x) for x in closes[symbol].drop_nulls().to_list()]
            if len(prices) < self._window:
                continue
            average = sum(prices) / len(prices)
            if average <= 0.0:
                continue
            deviation = (prices[-1] - average) / average
            scores.append((symbol, deviation))

        scores.sort(key=lambda x: x[1])
        picks = [s for s, _ in scores[: self._top_n]]
        if not picks:
            return Signal(information_available_at=stamp, weights={})

        weight = 1.0 / len(picks)
        return Signal(
            information_available_at=stamp,
            weights={s: weight for s in picks},
        )


def _latest_visible(view: MarketView) -> date:
    visible = view.history()
    if visible.is_empty():
        return date(1900, 1, 1)
    newest = visible["session_date"].max()
    assert isinstance(newest, date)
    return newest
```
