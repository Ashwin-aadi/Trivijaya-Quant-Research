## cross-sectional momentum

```python
from __future__ import annotations

from datetime import date

import polars as pl
from src.backtest.strategy import MarketView, Signal, Strategy


class CrossSectionalMomentum(Strategy):
    rationale = (
        "Stocks with the strongest medium-term relative performance often continue to "
        "outperform because information diffuses gradually and institutional investors "
        "adjust positions over time. Ranking securities cross-sectionally emphasizes "
        "relative strength while remaining diversified across the investable universe."
    )

    def __init__(self, lookback: int = 126, top_n: int = 10) -> None:
        self._lookback = lookback
        self._top_n = top_n

    def generate(self, view: MarketView) -> Signal:
        stamp = _latest_visible(view)
        closes = view.closes(lookback=self._lookback)
        if closes.height < self._lookback:
            return Signal(information_available_at=stamp, weights={})

        ranked: list[tuple[str, float]] = []
        for symbol in view.symbols:
            if symbol not in closes.columns:
                continue
            values = [float(v) for v in closes[symbol].drop_nulls().to_list()]
            if len(values) < self._lookback:
                continue
            if values[0] <= 0.0:
                continue
            ret = values[-1] / values[0] - 1.0
            ranked.append((symbol, ret))

        if not ranked:
            return Signal(information_available_at=stamp, weights={})

        ranked.sort(key=lambda x: x[1], reverse=True)
        picks = [s for s, _ in ranked[: self._top_n]]
        weight = 1.0 / len(picks)
        return Signal(
            information_available_at=stamp,
            weights={symbol: weight for symbol in picks},
        )


def _latest_visible(view: MarketView) -> date:
    visible = view.history()
    if visible.is_empty():
        return date(1900, 1, 1)
    newest = visible["session_date"].max()
    assert isinstance(newest, date)
    return newest
```

## volatility-scaled trend following

```python
from __future__ import annotations

from datetime import date

import polars as pl
from src.backtest.strategy import MarketView, Signal, Strategy


class VolatilityScaledTrendFollowing(Strategy):
    rationale = (
        "Persistent trends may arise because investors update expectations gradually. "
        "Scaling exposure inversely with realized volatility prevents the noisiest stocks "
        "from dominating portfolio risk while preserving exposure to positive trends."
    )

    def __init__(
        self,
        lookback: int = 60,
        vol_window: int = 20,
        max_weight: float = 0.10,
    ) -> None:
        self._lookback = lookback
        self._vol_window = vol_window
        self._max_weight = max_weight

    def generate(self, view: MarketView) -> Signal:
        needed = max(self._lookback, self._vol_window + 1)
        stamp = _latest_visible(view)
        closes = view.closes(lookback=needed)
        if closes.height < needed:
            return Signal(information_available_at=stamp, weights={})

        weights: dict[str, float] = {}
        for symbol in view.symbols:
            if symbol not in closes.columns:
                continue
            values = [float(v) for v in closes[symbol].drop_nulls().to_list()]
            if len(values) < needed:
                continue
            trend = values[-1] / values[-self._lookback] - 1.0
            if trend <= 0.0:
                continue

            returns = []
            for i in range(len(values) - self._vol_window, len(values)):
                prev = values[i - 1]
                if prev <= 0.0:
                    returns = []
                    break
                returns.append(values[i] / prev - 1.0)

            if len(returns) != self._vol_window:
                continue

            mean = sum(returns) / len(returns)
            var = sum((r - mean) ** 2 for r in returns) / len(returns)
            vol = var ** 0.5
            if vol <= 1e-9:
                continue

            weight = min(self._max_weight, trend / vol * 0.02)
            if weight > 0.0:
                weights[symbol] = weight

        return Signal(information_available_at=stamp, weights=weights)


def _latest_visible(view: MarketView) -> date:
    visible = view.history()
    if visible.is_empty():
        return date(1900, 1, 1)
    newest = visible["session_date"].max()
    assert isinstance(newest, date)
    return newest
```

## liquidity-screened equal weighting

```python
from __future__ import annotations

from datetime import date

import polars as pl
from src.backtest.strategy import MarketView, Signal, Strategy


class LiquidityScreenedEqualWeight(Strategy):
    rationale = (
        "Highly liquid stocks generally have lower trading costs and can absorb portfolio "
        "rebalancing more efficiently. Equal weighting across the most liquid names avoids "
        "concentration while retaining broad exposure to large-cap equities."
    )

    def __init__(self, lookback: int = 20, top_n: int = 20) -> None:
        self._lookback = lookback
        self._top_n = top_n

    def generate(self, view: MarketView) -> Signal:
        stamp = _latest_visible(view)
        history = view.history(lookback=self._lookback)
        if history.is_empty():
            return Signal(information_available_at=stamp, weights={})

        liquidity: list[tuple[str, float]] = []
        for symbol in view.symbols:
            rows = history.filter(pl.col("symbol") == symbol)
            if rows.height < self._lookback:
                continue
            closes = [float(v) for v in rows["adj_close"].to_list()]
            volumes = [float(v) for v in rows["volume"].to_list()]
            dollar = [c * v for c, v in zip(closes, volumes)]
            liquidity.append((symbol, sum(dollar) / len(dollar)))

        if not liquidity:
            return Signal(information_available_at=stamp, weights={})

        liquidity.sort(key=lambda x: x[1], reverse=True)
        picks = [s for s, _ in liquidity[: self._top_n]]
        weight = 1.0 / len(picks)
        return Signal(
            information_available_at=stamp,
            weights={symbol: weight for symbol in picks},
        )


def _latest_visible(view: MarketView) -> date:
    visible = view.history()
    if visible.is_empty():
        return date(1900, 1, 1)
    newest = visible["session_date"].max()
    assert isinstance(newest, date)
    return newest
```

## price-level reversion against a trailing reference

```python
from __future__ import annotations

from datetime import date

import polars as pl
from src.backtest.strategy import MarketView, Signal, Strategy


class PriceLevelTrailingReferenceReversion(Strategy):
    rationale = (
        "Prices that trade well below a long-run reference level may reflect temporary "
        "overshooting rather than permanent impairment. Buying the largest discounts to a "
        "trailing average seeks to benefit as prices revert toward their longer-term anchor."
    )

    def __init__(self, window: int = 60, top_n: int = 10) -> None:
        self._window = window
        self._top_n = top_n

    def generate(self, view: MarketView) -> Signal:
        stamp = _latest_visible(view)
        closes = view.closes(lookback=self._window)
        if closes.height < self._window:
            return Signal(information_available_at=stamp, weights={})

        scores: list[tuple[str, float]] = []
        for symbol in view.symbols:
            if symbol not in closes.columns:
                continue
            values = [float(v) for v in closes[symbol].drop_nulls().to_list()]
            if len(values) < self._window:
                continue
            reference = sum(values) / len(values)
            if reference <= 0.0:
                continue
            deviation = (values[-1] - reference) / reference
            scores.append((symbol, deviation))

        if not scores:
            return Signal(information_available_at=stamp, weights={})

        scores.sort(key=lambda x: x[1])
        picks = [s for s, _ in scores[: self._top_n]]
        weight = 1.0 / len(picks)
        return Signal(
            information_available_at=stamp,
            weights={symbol: weight for symbol in picks},

    newest = visible["session_date"].max()

    return newest
```
